# Kit · page script

A complete page script with two example charts, four summary tiles and a period selector. With [page.html.md](page.html.md), [styles.css.md](styles.css.md), [ui.js.md](ui.js.md), the chart library and the data file below, it is a working page. Build that page first, see it run, then replace the example parts.

The sections **format**, **theme**, **cards** and **start** come from `site/app.js` at commit `0440132`, with comments and two button labels translated. The reference site keeps a copy of them in each of its five page scripts, so every page is one self-contained file. Do the same: copy this file once per page.

Numbers and dates are formatted for `es-AR`, the reference locale, to match `ui.js` as shipped. [05-shell.md](../05-shell.md) lists the places to change together.

For the first run, save the two blocks below as `data.js` and `page.js` beside `index.html`. Each real page then gets its own pair, `<page>-data.js` and `<page>.js`, and its bundle sets a global of its own (`window.DATA`, `window.HISTORY`, …) so that a page can load several.

## `page.js`

```js
// One page of the site. It draws everything from window.DATA, which build.py writes to data.js.
// All real arithmetic lives in the pipeline; this file only formats, aggregates for display and draws.
// The sections "format", "theme", "cards" and "start" are the reference site's code: keep them.
// The sections marked EXAMPLE are placeholders: replace them with the page's own data and charts.
(() => {
  "use strict";

  const D = window.DATA;

  // EXAMPLE. A fixed order and a fixed colour slot per entity: the colour follows the entity, never the ranking.
  const ENTITIES = ["raw", "processed", "byproducts"];
  const ENAME = { raw: "Raw", processed: "Processed", byproducts: "By-products" };
  const ESLOT = { raw: "s1", processed: "s2", byproducts: "s3" };

  // ------------------------------------------------------------ format

  const nf = {};
  const num = (v, d = 0) => {
    if (v == null || Number.isNaN(v)) return "–";
    nf[d] ??= new Intl.NumberFormat("es-AR", { minimumFractionDigits: d, maximumFractionDigits: d });
    return nf[d].format(v);
  };
  const tons = (v) => `${num(v, Math.abs(v) < 100 ? 1 : 0)} t`;
  const usdM = (v) => `USD ${num(v / 1e6, 1)} M`;
  const perKg = (v) => (v == null ? "–" : `${num(v, 2)} USD/kg`);
  const pct = (v, d = 1) => `${num(v * 100, d)}%`;
  const signed = (v, d = 1) => `${v > 0 ? "+" : v < 0 ? "−" : ""}${num(Math.abs(v) * 100, d)}%`;
  const longDate = (iso) =>
    new Date(`${iso}T12:00:00`).toLocaleDateString("es-AR", { day: "numeric", month: "long", year: "numeric" });
  const sum = (xs) => xs.reduce((a, b) => a + (b || 0), 0);
  const esc = (s) => String(s).replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" })[c]);

  // ------------------------------------------------------------ theme

  let C = {};
  function readTokens() {
    const s = getComputedStyle(document.documentElement);
    const v = (n) => s.getPropertyValue(`--${n}`).trim();
    C = {
      page: v("page"), surface: v("surface"), ink: v("ink"), ink2: v("ink-2"), muted: v("muted"),
      grid: v("grid"), axis: v("axis"), border: v("border"), past: v("past"), empty: v("empty"),
      s1: v("s1"), s2: v("s2"), s3: v("s3"), s4: v("s4"), s5: v("s5"),
      seq: [v("seq-1"), v("seq-2"), v("seq-3"), v("seq-4"), v("seq-5")],
      ord1: v("ord-1"), ord2: v("ord-2"), ord3: v("ord-3"),
    };
  }
  const eColor = (k) => C[ESLOT[k]];

  // Text on top of a fill: white or near-black, by the luminance of the fill.
  function inkOn(hex) {
    const n = parseInt(hex.slice(1), 16);
    const lin = (c) => ((c /= 255) <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4);
    const L = 0.2126 * lin((n >> 16) & 255) + 0.7152 * lin((n >> 8) & 255) + 0.0722 * lin(n & 255);
    return L > 0.33 ? "#0b0b0b" : "#ffffff";
  }

  const swatch = (c, line = false) =>
    `<i style="display:inline-block;width:${line ? 12 : 8}px;height:${line ? 2 : 8}px;border-radius:2px;background:${c};margin:0 6px 2px 0;vertical-align:middle"></i>`;

  function base() {
    return {
      animationDuration: 400,
      textStyle: { fontFamily: 'system-ui, -apple-system, "Segoe UI", sans-serif', color: C.ink2 },
      grid: { left: 8, right: 24, top: 24, bottom: 8, containLabel: true },
      tooltip: {
        backgroundColor: C.surface,
        borderColor: C.axis,
        borderWidth: 1,
        textStyle: { color: C.ink, fontSize: 12 },
        extraCssText: "border-radius:8px;box-shadow:0 4px 16px rgba(0,0,0,.12);",
        confine: true,
      },
    };
  }
  const axisCat = (data, extra = {}) => ({
    type: "category",
    data,
    axisLine: { lineStyle: { color: C.axis } },
    axisTick: { show: false },
    axisLabel: { color: C.muted, fontSize: 11 },
    ...extra,
  });
  const axisVal = (fmt, extra = {}) => ({
    type: "value",
    axisLine: { show: false },
    axisTick: { show: false },
    axisLabel: { color: C.muted, fontSize: 11, formatter: fmt },
    splitLine: { lineStyle: { color: C.grid, type: "solid", width: 1 } },
    ...extra,
  });
  const barStyle = (color, top = true, horizontal = false) => ({
    color,
    borderColor: C.surface,
    borderWidth: 1,
    borderRadius: top ? (horizontal ? [0, 4, 4, 0] : [4, 4, 0, 0]) : 0,
  });
  // Stacked bars: only the last segment present in each bar gets the rounded end.
  function stacked(keys, rows, get, horizontal = false) {
    const last = rows.map((r) => [...keys].reverse().find((k) => get(r, k)));
    return keys.map((k) =>
      rows.map((r, i) => {
        const v = get(r, k);
        if (!v) return null;
        return last[i] === k
          ? { value: v, itemStyle: { borderRadius: horizontal ? [0, 4, 4, 0] : [4, 4, 0, 0] } }
          : { value: v };
      }),
    );
  }
  // An invisible series at the end of a stack, to print the total at the tip of each bar.
  const totalLabel = (stack, n, formatter) => ({
    type: "bar",
    stack,
    data: Array(n).fill(0),
    itemStyle: { color: "transparent" },
    label: { show: true, position: "right", color: C.ink2, fontSize: 11, formatter },
    tooltip: { show: false },
    silent: true,
  });

  const lineSeries = (name, data, color, extra = {}) => ({
    name,
    type: "line",
    data,
    symbol: "circle",
    symbolSize: 8,
    showSymbol: true,
    lineStyle: { width: 2, color, cap: "round", join: "round" },
    itemStyle: { color, borderColor: C.surface, borderWidth: 2 },
    emphasis: { focus: "series", scale: 1.4 },
    ...extra,
  });

  // ------------------------------------------------------------ supporting data (EXAMPLE)

  const months = D.months;
  // "season" in the names below means the period picked in the selector.
  const seasons = D.periods;
  let season = seasons[seasons.length - 1].id;
  const S = () => seasons.find((s) => s.id === season);
  const seasonLabel = (s) => s.label;

  // ------------------------------------------------------------ charts (EXAMPLE)

  const SPECS = {
    monthly: {
      title: "Tons shipped per month",
      sub: () => `${months[0].label} to ${months[months.length - 1].label}, by product`,
      legend: () => ENTITIES.map((k) => ({ label: ENAME[k], color: eColor(k) })),
      foot: "Invented figures, to show the card. A real footnote names the source table and what to watch for.",
      option() {
        const data = stacked(ENTITIES, months, (m, k) => m.tons[k]);
        return {
          ...base(),
          tooltip: {
            ...base().tooltip,
            trigger: "axis",
            axisPointer: { type: "shadow", shadowStyle: { color: C.grid, opacity: 0.5 } },
            formatter: (ps) => {
              const m = months[ps[0].dataIndex];
              const rows = ENTITIES.filter((k) => m.tons[k])
                .map((k) => `${swatch(eColor(k))}${ENAME[k]}: <b>${tons(m.tons[k])}</b>`)
                .join("<br>");
              return `<b>${esc(m.label)}</b><br>${rows}<br>Total: <b>${tons(m.total_t)}</b>`;
            },
          },
          xAxis: axisCat(months.map((m) => m.label)),
          yAxis: axisVal((v) => num(v)),
          series: ENTITIES.map((k, i) => ({
            name: ENAME[k],
            type: "bar",
            stack: "t",
            barMaxWidth: 24,
            data: data[i],
            itemStyle: barStyle(eColor(k), false),
            emphasis: { focus: "series" },
          })),
        };
      },
      table: () => ({
        head: ["Month", ...ENTITIES.map((k) => ENAME[k]), "Total t"],
        rows: months.map((m) => [
          m.label,
          ...ENTITIES.map((k) => (m.tons[k] == null ? "" : num(m.tons[k], 1))),
          num(m.total_t, 1),
        ]),
      }),
    },

    destinations: {
      seasonal: true,
      title: "Destinations",
      sub: () => `Tons, ${seasonLabel(S())}`,
      height: "short",
      option() {
        // A horizontal bar chart draws its first category at the bottom: sort ascending to put the largest on top.
        const rows = [...S().destinations].sort((a, b) => a.tons - b.tons);
        return {
          ...base(),
          grid: { left: 8, right: 72, top: 8, bottom: 8, containLabel: true },
          tooltip: {
            ...base().tooltip,
            formatter: (p) => `<b>${esc(p.name)}</b><br>${tons(p.value)} · ${pct(p.value / S().tons)} of the total`,
          },
          xAxis: axisVal((v) => num(v)),
          yAxis: axisCat(rows.map((d) => d.name), { axisLabel: { color: C.ink2, fontSize: 12 } }),
          series: [
            {
              type: "bar",
              barMaxWidth: 20,
              data: rows.map((d) => d.tons),
              itemStyle: barStyle(C.s1, true, true),
              label: { show: true, position: "right", color: C.ink2, fontSize: 11, formatter: (p) => tons(p.value) },
            },
          ],
        };
      },
      table: () => ({
        head: ["Destination", "Tons", "Share"],
        rows: [...S().destinations]
          .sort((a, b) => b.tons - a.tons)
          .map((d) => [d.name, num(d.tons, 1), pct(d.tons / S().tons)]),
      }),
    },
  };

  // ------------------------------------------------------------ cards

  const instances = {};

  function buildCard(el, id) {
    const spec = SPECS[id];
    el.innerHTML = `
      <div class="card-head">
        <div><p class="card-title">${spec.title}</p><p class="card-sub"></p></div>
        <button class="btn" type="button" aria-expanded="false">Show table</button>
      </div>
      <div class="chart ${spec.height || ""}" role="img" aria-label="${esc(spec.title)}"></div>
      <div class="legend"></div>
      <div class="table-wrap" hidden></div>
      <p class="card-foot"></p>`;
    el.querySelector(".legend").addEventListener("click", (e) => {
      const b = e.target.closest("button[data-series]");
      if (!b) return;
      b.setAttribute("aria-pressed", String(b.getAttribute("aria-pressed") !== "true"));
      instances[id]?.dispatchAction({ type: "legendToggleSelect", name: b.dataset.series });
    });
    const btn = el.querySelector(".btn");
    const wrap = el.querySelector(".table-wrap");
    btn.addEventListener("click", () => {
      const open = wrap.hidden;
      wrap.hidden = !open;
      btn.textContent = open ? "Hide table" : "Show table";
      btn.setAttribute("aria-expanded", String(open));
      if (open) renderTable(id);
    });
  }

  function renderTable(id) {
    const el = document.querySelector(`[data-chart="${id}"] .table-wrap`);
    if (el.hidden) return;
    const t = SPECS[id].table();
    el.innerHTML = `<table><thead><tr>${t.head.map((h) => `<th>${esc(h)}</th>`).join("")}</tr></thead><tbody>${t.rows
      .map((r) => `<tr>${r.map((c) => `<td>${esc(c)}</td>`).join("")}</tr>`)
      .join("")}</tbody></table>`;
  }

  function render(id) {
    const card = document.querySelector(`[data-chart="${id}"]`);
    const spec = SPECS[id];
    card.querySelector(".card-sub").textContent = spec.sub ? spec.sub() : "";
    const foot = typeof spec.foot === "function" ? spec.foot() : spec.foot || "";
    const footEl = card.querySelector(".card-foot");
    footEl.textContent = foot;
    footEl.hidden = !foot;
    const legend = spec.legend ? spec.legend() : [];
    card.querySelector(".legend").innerHTML = legend
      .map((l) => {
        const inner = `<i class="${l.line ? "line" : ""}" style="background:${l.color}"></i>${esc(l.label)}`;
        return l.series
          ? `<button type="button" class="leg-btn" aria-pressed="true" data-series="${esc(l.series)}">${inner}</button>`
          : `<span>${inner}</span>`;
      })
      .join("");
    const node = card.querySelector(".chart");
    instances[id] ??= echarts.init(node, null, { renderer: "canvas" });
    const option = spec.option();
    // With "reduce motion" on, the chart appears without animation.
    if (matchMedia("(prefers-reduced-motion: reduce)").matches) option.animation = false;
    instances[id].setOption(option, true);
    renderTable(id);
  }

  // ------------------------------------------------------------ texts (EXAMPLE)

  function renderKpis() {
    const c = D.comparison;
    const a = c.current;
    const b = c.previous;
    const s = seasons[seasons.length - 1];
    const lead = [...s.destinations].sort((x, y) => y.tons - x.tons)[0];
    const delta = (x, y) => {
      const d = x / y - 1;
      return `<span class="${d >= 0 ? "up" : "down"}">${d >= 0 ? "▲" : "▼"} ${signed(d)}</span> against ${esc(b.label)}`;
    };
    document.getElementById("kpis").innerHTML = `
      <div class="tile hero">
        <div class="label">Implied price, ${esc(a.label)}</div>
        <div class="value">${num(a.usd_kg, 2)}<span class="unit">USD/kg</span></div>
        <div class="delta">${delta(a.usd_kg, b.usd_kg)} (${num(b.usd_kg, 2)})</div>
      </div>
      <div class="tile">
        <div class="label">Tons shipped</div>
        <div class="value">${num(a.tons, 1)}<span class="unit">t</span></div>
        <div class="delta">${delta(a.tons, b.tons)} (${num(b.tons)} t)</div>
      </div>
      <div class="tile">
        <div class="label">Revenue</div>
        <div class="value">${num(a.usd / 1e6, 1)}<span class="unit">M USD</span></div>
        <div class="delta">${delta(a.usd, b.usd)} (${num(b.usd / 1e6, 1)} M)</div>
      </div>
      <div class="tile">
        <div class="label">Leading destination, ${esc(s.id)}</div>
        <div class="value">${esc(lead.name)}</div>
        <div class="delta">${pct(lead.tons / s.tons)} of the tons · ${num(lead.tons)} t</div>
      </div>`;
    document.getElementById("kpi-note").textContent =
      `Current period (${a.label}) against the same ${c.months} months of the previous one.`;
  }

  function renderTexts() {
    document.getElementById("sub").textContent =
      `Example publisher · ${months.length} months · data to ${months[months.length - 1].label}`;
    document.getElementById("foot").textContent =
      `Data generated on ${longDate(D.generated)} by build.py from the stored source files.`;
  }

  // ------------------------------------------------------------ period selector

  // Arrow keys, Home and End pick another period, as in a group of radio buttons.
  function arrowSelect(e, seg) {
    const bs = [...seg.querySelectorAll("button")];
    const i = bs.indexOf(e.currentTarget);
    const j = { ArrowRight: i + 1, ArrowDown: i + 1, ArrowLeft: i - 1, ArrowUp: i - 1, Home: 0, End: bs.length - 1 }[e.key];
    if (j === undefined) return;
    e.preventDefault();
    bs[(j + bs.length) % bs.length].click();
  }

  function renderSeasonSeg() {
    const seg = document.getElementById("season-seg");
    seg.innerHTML = [...seasons]
      .reverse()
      .map(
        (s) =>
          `<button type="button" role="radio" aria-checked="${s.id === season}" tabindex="${s.id === season ? 0 : -1}" data-s="${esc(s.id)}">${esc(seasonLabel(s))}</button>`,
      )
      .join("");
    seg.querySelectorAll("button").forEach((b) => {
      b.addEventListener("click", () => {
        season = b.dataset.s;
        renderSeasonSeg();
        seg.querySelector('[aria-checked="true"]').focus(); // the selector is redrawn: without this the keyboard loses focus
        renderSeasonal();
      });
      b.addEventListener("keydown", (e) => arrowSelect(e, seg));
    });
  }

  // EXAMPLE. Everything that depends on the picked period: its lede and the charts flagged `seasonal`.
  function renderSeasonal() {
    const s = S();
    const top = [...s.destinations].sort((a, b) => b.tons - a.tons)[0];
    document.getElementById("destinations-lede").textContent =
      `${seasonLabel(s)}: ${s.destinations.length} destinations. ${top.name} takes ${pct(top.tons / s.tons, 0)} of the tons.`;
    for (const id of Object.keys(SPECS)) if (SPECS[id].seasonal) render(id);
  }

  function renderAll() {
    readTokens();
    renderKpis();
    for (const id of Object.keys(SPECS)) render(id);
    renderSeasonal();
  }

  // ------------------------------------------------------------ start

  document.querySelectorAll("[data-chart]").forEach((el) => buildCard(el, el.dataset.chart));
  renderTexts();
  renderSeasonSeg();
  renderAll();

  // The period selector stays fixed just under the header, whose height changes on narrow screens.
  function placeSeasonBar() {
    const h = document.querySelector(".top").offsetHeight;
    document.getElementById("season-bar").style.top = `${h + 8}px`;
    document.documentElement.style.scrollPaddingTop = `${h + 70}px`;
  }
  placeSeasonBar();

  window.matchMedia("(prefers-color-scheme: dark)").addEventListener("change", renderAll);
  let pending;
  window.addEventListener("resize", () => {
    clearTimeout(pending);
    pending = setTimeout(() => {
      placeSeasonBar();
      Object.values(instances).forEach((i) => i.resize());
    }, 120);
  });
})();
```
